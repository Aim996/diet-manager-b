  created_at = (Get-Date).ToString('o')
  snapshot_commit = $snapshotCommit
  source_commit = 'e46ebccbb741bab3ac3b324bdb5531bd6032a01f'
  build_commit = 'a723c25f3c28a45f842aea0ed6c29240c503aa51'

  Set-Content -Encoding utf8NoBOM -LiteralPath "$releaseRoot\SHA256SUMS.txt"
